## Program 1: Basic Key-Value Operations

```python
import requests

url = "http://localhost:8098/types/default/buckets/students/keys/101"

data = {"name": "Rahul", "course": "BCA"}

# Store
print(requests.put(url, json=data).status_code)

# Retrieve
print(requests.get(url).json())

# Update
data["name"] = "Aman"
requests.put(url, json=data)
print(requests.get(url).json())

# Delete
print(requests.delete(url).status_code)
```

## Program 5: Structure of Data in Key-Value Store

```python
import requests

base = "http://localhost:8098/types/default/buckets/data/keys/"

values = {
    "string": "Hello Riak",
    "json": {"name": "Aman", "age": 20},
    "list": ["Python", "Java", "Dart"]
}

for key, value in values.items():
    requests.put(base + key, json=value)

for key in values:
    print(key, requests.get(base + key).json())

# Bucket = collection
# Key = unique identifier
# Value = stored data
```

## Program 7: Scaling in Riak KV

```python
import requests

buckets = ["products1", "products2", "products3"]

for bucket in buckets:
    for i in range(1, 4):
        url = (
            "http://localhost:8098/types/default/"
            f"buckets/{bucket}/keys/product{i}"
        )

        data = {
            "id": i,
            "name": f"Product {i}",
            "price": i * 100
        }

        response = requests.put(url, json=data)
        print(bucket, i, response.status_code)

print("Data inserted into multiple buckets.")
```

## Program 8: Multi-Operation Shopping Cart

```python
import requests

url = "http://localhost:8098/types/default/buckets/carts/keys/cart1"

cart = {"items": [], "total": 0}

cart["items"].append(
    {"name": "Book", "price": 200, "qty": 2}
)
cart["items"].append(
    {"name": "Pen", "price": 10, "qty": 3}
)

cart["total"] = sum(
    item["price"] * item["qty"]
    for item in cart["items"]
)

requests.put(url, json=cart)
print("Cart:", requests.get(url).json())

cart["items"][0]["qty"] = 3

cart["total"] = sum(
    item["price"] * item["qty"]
    for item in cart["items"]
)

requests.put(url, json=cart)
print("Updated:", requests.get(url).json())

print("Riak KV does not provide general multi-key ACID transactions.")
```

## Program 9: Query-by-Data Functionality

```python
import requests

base = "http://localhost:8098/types/default/buckets/users/keys/"

users = [
    {"id": 1, "name": "Aman", "city": "Agra", "age": 20},
    {"id": 2, "name": "Riya", "city": "Delhi", "age": 22},
    {"id": 3, "name": "Ravi", "city": "Agra", "age": 25}
]

for user in users:
    requests.put(base + str(user["id"]), json=user)

for user in users:
    response = requests.get(base + str(user["id"]))

    if response.ok:
        data = response.json()

        if data["city"] == "Agra":
            print(data)

print("Query completed.")
```

## Program 10: Set Operations

```python
import requests

base = "http://localhost:8098/types/default/buckets/interests/keys/"

set1 = {"Python", "Java", "Dart"}
set2 = {"Java", "C++", "Python"}

requests.put(base + "user1", json=sorted(set1))
requests.put(base + "user2", json=sorted(set2))

set1.add("Flutter")
set1.discard("Dart")

print("Membership:", "Python" in set1)
print("Union:", set1 | set2)
print("Intersection:", set1 & set2)
print("Difference:", set1 - set2)
print("Updated Set:", set1)
```

## Program 11: University Graph

```cypher
CREATE (s:Student {
    student_id: 101,
    name: "Rahul"
})

CREATE (c:Course {
    course_id: 1,
    name: "BCA"
})

    department_id: 1,
    name: "Computer Applications"
})
```

## Program 15: Graph Database Application Using Python

Install the driver:

```bash
pip install neo4j
```

Python code:

```python
from neo4j import GraphDatabase

URI = "bolt://localhost:7687"
USERNAME = "neo4j"
PASSWORD = "your_password"

driver = GraphDatabase.driver(
    URI,
    auth=(USERNAME, PASSWORD)
)


def create_person(name, city):
    with driver.session() as session:
        session.run(
            "CREATE (p:Person {name: $name, city: $city})",
            name=name,
            city=city
        )
        print("Person created.")


def display_people():
    with driver.session() as session:
        result = session.run(
            "MATCH (p:Person) "
            "RETURN p.name AS name, p.city AS city"
        )

        for record in result:
            print(record["name"], record["city"])


def update_person(name, city):
    with driver.session() as session:
        session.run(
            "MATCH (p:Person {name: $name}) "
            "SET p.city = $city",
            name=name,
            city=city
        )
        print("Person updated.")


def delete_person(name):
    with driver.session() as session:
        session.run(
            "MATCH (p:Person {name: $name}) "
            "DETACH DELETE p",
            name=name
        )
        print("Person deleted.")


try:
    driver.verify_connectivity()

    create_person("Rahul", "Agra")
    create_person("Aman", "Delhi")

    print("All People:")
    display_people()

    update_person("Rahul", "Noida")

    print("After Update:")
    display_people()

    delete_person("Aman")

    print("After Delete:")
    display_people()

finally:
    driver.close()
```

Change `your_password` to your Neo4j database password before running the program.

## Extra queries for Program 15

```cypher
// Social network
MATCH (p:Person)-[:FRIENDS_WITH]-(friend:Person)
RETURN p.name, friend.name;

// Product recommendation
MATCH (p:Product)-[:BELONGS_TO]->(c:Category)
      <-[:BELONGS_TO]-(other:Product)
WHERE p <> other
RETURN p.name, other.name;

// Fraud detection pattern
MATCH (a:Customer)-[:USES_CONTACT]->(contact)
      <-[:USES_CONTACT]-(b:Customer)
WHERE a <> b
RETURN a.name, b.name;

// Knowledge graph
MATCH (a)-[r]->(b)
RETURN a, r, b
LIMIT 20;
```
